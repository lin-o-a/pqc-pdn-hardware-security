void print_hex(const char *label, const uint8_t *data, size_t len) {
    printf("%s (first %d bytes):\r\n", label, (int)len);
    for (size_t i = 0; i < len; i++) {
        printf("%02X ", data[i]);
        if ((i + 1) % 16 == 0) printf("\r\n");
    }
    printf("\r\n");
    fflush(stdout);
}

  /* USER CODE BEGIN 2 */

  /* Initialize leds */
  BSP_LED_Init(LED_GREEN);
  BSP_LED_Init(LED_BLUE);
  BSP_LED_Init(LED_RED);

  /* 1. CHECK FOR DYNAMIC BOR RESET */
  if (__HAL_RCC_GET_FLAG(RCC_FLAG_BORRST) != RESET) {
      /* Dynamic BOR occurred! Rapidly blink GREEN + BLUE LEDs 10 times */
      for (int i = 0; i < 10; i++) {
          BSP_LED_Toggle(LED_GREEN);
          BSP_LED_Toggle(LED_BLUE);
          HAL_Delay(100);
      }

      /* Always clear flags after reading */
      __HAL_RCC_CLEAR_RESET_FLAGS();
  }

  /* 2. STEADY STATE INDICATOR */
  /* Turn GREEN + BLUE ON solid before entering heavy computation */
  BSP_LED_On(LED_GREEN);
  BSP_LED_On(LED_BLUE);

  uint8_t pk[PQCLEAN_MLKEM512_CLEAN_CRYPTO_PUBLICKEYBYTES];
  uint8_t sk[PQCLEAN_MLKEM512_CLEAN_CRYPTO_SECRETKEYBYTES];
  uint8_t ct[PQCLEAN_MLKEM512_CLEAN_CRYPTO_CIPHERTEXTBYTES];
  uint8_t ss_a[PQCLEAN_MLKEM512_CLEAN_CRYPTO_BYTES];
  uint8_t ss_b[PQCLEAN_MLKEM512_CLEAN_CRYPTO_BYTES];

  /* USER CODE BEGIN WHILE */
  while (1)
  {
    /* USER CODE END WHILE */
	  // Check if a BOR occurred during the previous loop iteration
	  if (__HAL_RCC_GET_FLAG(RCC_FLAG_BORRST) != RESET) {
		  BSP_LED_On(LED_RED);
		  printf("[ALERT] Dynamic BOR occurred inside execution loop!\r\n");
		  __HAL_RCC_CLEAR_RESET_FLAGS();
	  }

	  /* 1. Key Generation */
	  PQCLEAN_MLKEM512_CLEAN_crypto_kem_keypair(pk, sk);

	  /* 2. Encapsulation (Sender generates ciphertext & shared secret) */
	  PQCLEAN_MLKEM512_CLEAN_crypto_kem_enc(ct, ss_a, pk);

	  /* 3. Decapsulation (Receiver decodes ciphertext to get shared secret) */
	  PQCLEAN_MLKEM512_CLEAN_crypto_kem_dec(ss_b, ct, sk);
	  /* Current Surge Loop: Forces rapid transistor switching to drop rail voltage */
	  for (volatile uint32_t i = 0; i < 10000; i++) {
	      /* Toggle register bits violently to maximize dynamic current draw */
	      volatile uint32_t x = 0xAAAAAAAA;
	      x ^= 0x55555555;
	      x ^= 0xAAAAAAAA;
	      x ^= 0x55555555;
	  }
    /* USER CODE BEGIN 3 */
  }
  /* USER CODE END 3 */
