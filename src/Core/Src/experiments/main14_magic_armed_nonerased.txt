int _write(int file, char *ptr, int len) {
    for (int i = 0; i < len; i++) {
        // Send byte over ITM Port 0
        ITM_SendChar(*ptr++);
    }
    return len;
}

void print_hex(const char *label, const uint8_t *data, size_t len) {
    printf("%s (first %d bytes):\r\n", label, (int)len);
    for (size_t i = 0; i < len; i++) {
        printf("%02X ", data[i]);
        if ((i + 1) % 16 == 0) printf("\r\n");
    }
    printf("\r\n");
    fflush(stdout);
}

/**
  * @brief  The application entry point.
  * @retval int
  */
int main(void)
{
  HAL_Init();
  SystemClock_Config();
  MX_GPIO_Init();
  MX_ICACHE_Init();

  /* USER CODE BEGIN 2 */
  BSP_LED_Init(LED_GREEN);
  BSP_LED_Init(LED_BLUE);
  BSP_LED_Init(LED_RED);

  // Clear all LEDs at startup
  BSP_LED_Off(LED_GREEN);
  BSP_LED_Off(LED_BLUE);
  BSP_LED_Off(LED_RED);

  uint8_t bor_occurred = (__HAL_RCC_GET_FLAG(RCC_FLAG_BORRST) != RESET);

    // 1. DEMONSTRATE SECURITY LEAK ON BOR RECOVERY
    if (bor_occurred) {
    	__HAL_RCC_CLEAR_RESET_FLAGS();

        // If RAM persisted and contains key material despite the reboot:
        if (persistent_magic_flag == MAGIC_ARMED) {
            // SECURITY COMPROMISED: MCU rebooted via BOR, but sensitive key
            // material survived in SRAM without zeroization!
            BSP_LED_On(LED_RED);
            BSP_LED_On(LED_BLUE); // PURPLE/RED+BLUE = Key Leak Captured!
            //while(1);
        }
    }

    // 2. PRE-ARM RAM BEFORE DANGEROUS HIGH-CURRENT WORKLOAD
      // Fill RAM with a dummy secret key pattern (or partial key) & set flag
      /*for (int i = 0; i < 32; i++) {
          persistent_secret_key[i] = 0xEE; // Test key pattern
      }*/
      persistent_magic_flag = MAGIC_ARMED;


  uint8_t pk[PQCLEAN_MLKEM512_CLEAN_CRYPTO_PUBLICKEYBYTES];
  uint8_t sk[PQCLEAN_MLKEM512_CLEAN_CRYPTO_SECRETKEYBYTES];
  uint8_t ct[PQCLEAN_MLKEM512_CLEAN_CRYPTO_CIPHERTEXTBYTES];
  uint8_t ss_a[PQCLEAN_MLKEM512_CLEAN_CRYPTO_BYTES];
  uint8_t ss_b[PQCLEAN_MLKEM512_CLEAN_CRYPTO_BYTES];

  /* USER CODE BEGIN WHILE */
  while (1)
  {
	  /* 1. Key Generation */
	  PQCLEAN_MLKEM512_CLEAN_crypto_kem_keypair(pk, sk);
	  /* 2. Encapsulation */
	  PQCLEAN_MLKEM512_CLEAN_crypto_kem_enc(ct, ss_a, pk);
	  /* 2. Decapsulation */
	  PQCLEAN_MLKEM512_CLEAN_crypto_kem_dec(ss_b, ct, sk);

	  // Normal continuous execution heartbeat
	  BSP_LED_Toggle(LED_GREEN);
	  HAL_Delay(200);
  }
}
